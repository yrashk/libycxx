# The names a plain-std block may declare (`namespace [[__gnu__::__visibility__("hidden")]] std {
# // plain std (DECISIONS §20.5)`): what the compilers or the platform name in ::std itself rather
# than in the inline ABI namespace std::__y1. Each is hidden in every mode. Found by experiment
# (docs/design/shared-probes on the shared-lib-design branch; tests/ycxx/conformance/abi_namespace).

destroying_delete               # its tag object, kept with it

                                # sanitizers intercept and src/runtime/new hides by name
initializer_list                # GCC deduces from braced lists only to it
byte to_integer operator        # GCC's may-alias rule for std::byte; its operators, with it
size_t ptrdiff_t nullptr_t max_align_t  # <cstddef>'s using-declarations, in std::byte's block
terminate                       # Clang's __clang_call_terminate calls _ZSt9terminatev by name
meta                            # GCC (-freflection) predeclares namespace std::meta
