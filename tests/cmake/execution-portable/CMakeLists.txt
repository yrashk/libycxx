cmake_minimum_required(VERSION 3.28)
project(libycxx_execution_test LANGUAGES CXX)

enable_testing()
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(YCXX_TEST_EXECUTION "Test the C++26 portable execution provider" ON)
option(YCXX_TEST_NATIVE "Also test the installed libycxx provider" ON)
option(YCXX_TEST_PORTABLE_PACKAGE "Test the installed portable package" OFF)
if(YCXX_TEST_PORTABLE_PACKAGE)
  find_package(libycxxExecution CONFIG REQUIRED)
else()
  add_subdirectory(../../../portable portable EXCLUDE_FROM_ALL)
endif()

add_executable(portable_stop_token stop_token.cpp)
target_link_libraries(portable_stop_token PRIVATE ycxx::stop_token_portable)
target_compile_definitions(portable_stop_token PRIVATE YCXX_TEST_PORTABLE)
add_test(NAME portable_stop_token COMMAND portable_stop_token)
set_tests_properties(portable_stop_token PROPERTIES TIMEOUT 20)

if(YCXX_TEST_NATIVE)
  find_package(libycxx CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_executable(native_stop_token stop_token.cpp)
  target_link_libraries(native_stop_token PRIVATE ycxx::ycxx Threads::Threads)
  add_test(NAME native_stop_token COMMAND native_stop_token)
  set_tests_properties(native_stop_token PROPERTIES TIMEOUT 20)
endif()

if(YCXX_TEST_EXECUTION)
  add_executable(portable_execution execution.cpp)
  target_link_libraries(portable_execution PRIVATE ycxx::execution_portable)
  add_test(NAME portable_execution COMMAND portable_execution)
  set_tests_properties(portable_execution PROPERTIES TIMEOUT 20)
  if(YCXX_TEST_PORTABLE_PACKAGE)
    add_executable(portable_execution_module execution_module.cpp)
    set_target_properties(portable_execution_module PROPERTIES CXX_SCAN_FOR_MODULES ON)
    target_link_libraries(portable_execution_module PRIVATE ycxx::execution_portable_modules)
    add_test(NAME portable_execution_module COMMAND portable_execution_module)
    set_tests_properties(portable_execution_module PROPERTIES TIMEOUT 20)
  endif()
endif()

foreach(target IN ITEMS portable_stop_token native_stop_token portable_execution portable_execution_module)
  if(TARGET ${target})
    target_include_directories(${target} PRIVATE ../../ycxx/support)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
